# The boss fights. How to write these: docs/level-format.md.
#
# [boss N escorts] are the ships sent in, over and over, while boss N is on screen (from where the ship is when the
# previous batch is used up). [boss 1 hangar] is the six suicide ships boss 1 launches from its hangar each attack:
# their positions are on the same 12 x 12 grid, but measured from the boss art's top-left corner, not the screen's.

[boss 1 escorts]
@0:13 missile sn  0,4 7,6 12,6
+0.8 missile sn  12,3 5,5 0,5
repeat 0.8 0/1/2/3/4 2
@0:25 laser ss  0,9
+0.5 mirror
extreme+: +0.5 laser ss  0,6
mirror
insane+: +0.5 laser ss  0,3
mirror
@0:25.5 wait   # every difficulty's batch lasts as long as Hard's

[boss 1 hangar]
@0:00 ship msn  1.8,2.2  halt 4.11  attack
ship msn  2.5,2.66  halt 3.34  attack
normal+: ship msn  3,3.18  halt 2.57  attack
normal+: ship msn  6.92,3.18  halt 2.57  attack
ship msn  7.45,2.66  halt 3.34  attack
ship msn  8.14,2.2  halt 4.11  attack
