# Millennium Typer: tuning numbers
#
# Every line is "name = value". Anything after a # is a note for people and is ignored.
# The game checks this file when it starts: a typo or a missing line stops it with a message saying which line.
# With the editing keys on (--edit), F5 reloads this file along with the level.
#
# Units: positions and distances are pixels in the 600 x 550 game view. Times are ticks (24 a second) unless a line
# says seconds.
# "cps" is the difficulty's typing speed in characters per second (words per minute x 5 / 60): Easy 2.08 (25 WPM),
# Normal 3.33 (40), Hard 5.42 (65), Extreme 7.5 (90), Insane 10 (120). Speeds that say "x cps" are multiplied by
# it, so everything gets faster on harder difficulties, and times that say "/ cps" are divided by it. A number can
# be written as a fraction (100/3) when it has no exact decimal.
#
# These are the original game's numbers. Change them to retune; docs/level-format.md explains ship codes.

[enemy ships]
# Every enemy starts with these, then each letter of its ship code adds to them.
points = 5
speed = 0.5               # acceleration; plus what the ship's letters add below, then multiplied by cps x speed per cps
speed per cps = 0.12      # (wpm / 100 in the original)
pace above hard = 0.5     # Above Hard, everything that says "x cps" or "/ cps" uses Hard's cps x (cps / Hard's)^this instead,
                          #   so the hardest difficulties bring more ships rather than ships too fast to read: at 0.5
                          #   Extreme moves 1.18x as fast as Hard and Insane 1.36x (about what 75 and 90 WPM did).
                          #   1 = the full typing speed. The fire rate and scores always use the full cps.

[size]
# The ship code's 1st letter: word length, and points and speed added.
s = letters 4, points 0, speed 0          # small
m = letters 6, points 1, speed -0.2       # medium
l = letters 8, points 3, speed -0.25      # large
x = letters 7, points 3, speed -0.1       # stealth: fades in, always fires missiles ("xm")
n = letters 7, points 0, speed 0          # invisible laser turret (dropped by boss 2: "nl")

[weapon]
# The ship code's 2nd letter. m (missile) changes nothing.
laser extra letters = 1   # l: a laser ship's word is this much longer
laser points = 1
laser speed = -0.1
suicide letters = 3       # s: suicide ships and mines always have this many letters
suicide points = -3
suicide speed = 0.2
mine speed = -0.2         # used instead of suicide speed for small suicide ships (mines)

[defence]
# The ship code's 3rd letter: extra points.
n = points 0              # none
s = points 2              # shield: an EMP word has to knock it out first
t = points 2              # turret: shoots down player missiles
b = points 5              # both

[hull health]
# The ship code's first two letters: the hulls that exist, and how many player missiles each one takes.
sm = 1
sl = 1
ss = 1
mm = 2
ml = 2
ms = 1
lm = 4
ll = 4
xm = 1
nl = 1

[enemy weapons]
missile speed = 0.096     # x cps
missile points = 1        # for shooting down a missile (none during a boss fight)
fire rate cps = 12.5      # each fire command adds cps / 12.5 to a counter, kept between the min and max below;
fire rate min = 0.3       #   a missile only launches when the counter passes 1
fire rate max = 1
fire rate health 2 = 1.5  # ...times this for ships with 2 health
fire rate health 3 = 2    # ...and this for 3 or more
laser charge = 500        # a laser charges for this / cps ticks
turret range = 100        # turrets shoot down player missiles this close to their ship
stealth fade cps = 2.5    # stealth ships fade in over about 100 / (1 + cps / this) ticks

[player]
health = 5
typing bar = 20           # letters the typing bar holds; each letter past that costs a point
heal word shortest = 8    # the hull-repair word's length, picked once per game
heal word longest = 9
overflow = 30             # ticks a word stays typeable after its ship leaves the screen
missile speed = 1.2       # x cps
missile hit range = 30
emp charge = 5            # ticks before an EMP (typed at a shield) hits

[level scripts]
jitter = 10               # ships aim up to this far either side of each grid point (inside the view)
arrive distance = 20      # how close a ship gets to a point before moving on
halt scale = 100/3        # converts halts between difficulties: a halt of S seconds (as on Hard) waits S x 24 x 5.42 / cps
                          # ticks. The built-in ship moves (laser, stealth) use the original's halt units, N x this / cps.
end wait = 2.77           # seconds between the last ship leaving and the level ending
copy gap = 0.55           # seconds between a wave and a copy of it (for the coming difficulty pass; nothing uses it yet)

[bosses]
health = 3                # hits to kill a boss
points = 100              # for the kill
scanner ring = 300        # the scanner ring's starting radius
scanner step = 30         # each scanner word shrinks the ring this much
scanner word = 3          # letters in the scanner's words
scanner final word = 10   # letters in the word that hits the boss
first attack = 6          # halt before a boss's first attack (scaled like "halt")
boss 1 attack = 48        # halt between attacks: boss 1 launches six suicide ships
boss 2 attack = 23        # boss 2 drops an invisible laser turret...
boss 2 attack per health = 3   # ...waiting this much longer for each health it has left
boss 3 attack = 70        # boss 3 charges its beam
boss 3 beam = 25          # the beam's timing unit: on Hard a step is this / cps ticks, rounded down (4); other difficulties
                          #   scale Hard's step by Hard's cps / their typing speed, so the counter-words take the same
                          #   share of the speed everywhere. The whole attack lasts 44 steps
counter word = 11         # boss 3's counter-words have this many letters, less the boss's health
